from typing import List

class Solution:
    def fullJustify(self, words: List[str], maxWidth: int) -> List[str]:
        result = []
        i = 0
        n = len(words)

        while i < n:
            j = i
            line_length = 0

            while j < n and line_length + len(words[j]) + (j - i) <= maxWidth:
                line_length += len(words[j])
                j += 1

            num_words = j - i
            num_gaps = num_words - 1

            if j == n or num_gaps == 0:
                line = " ".join(words[i:j])
                line += " " * (maxWidth - len(line))

            else:
                total_spaces = maxWidth - line_length
                spaces_per_gap = total_spaces // num_gaps
                extra_spaces = total_spaces % num_gaps

                parts = []

                for k in range(num_gaps):
                    parts.append(words[i + k])

                    spaces = spaces_per_gap
                    if k < extra_spaces:
                        spaces += 1

                    parts.append(" " * spaces)

                parts.append(words[j - 1])
                line = "".join(parts)

            result.append(line)
            i = j

        return result
