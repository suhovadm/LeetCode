```
 # Импортируем List, чтобы указать, что переменная или результат - это список определённого типа.  
from typing import List  

# Заводим класс Solution, как того требует LeetCode.  
# Объявляем функцию fullJustify, которая форматирует текст.  
# На вход поступает список слов words: List[str]  
# maxWidth: int - максимальная ширина строки (целое число, т.к. int).  
# -> List[str] - это значит, что функция должна вернуть список строк.  
class Solution:  
    def fullJustify(self, words: List[str], maxWidth: int) -> List[str]:  
        
	# Заводим пустой список, в который мы будем складывать уже отформатированные строки.  
	result = []  
        # Индекс первого слова текущей строки, начиная с нуля.  
	i = 0  
        # Количество слов. len - считает слова.  
	n = len(words)  
	
	# Запускаем внешний цикл while. Говорим "пока условие истинно, повторяй действия".  
	# Условие i < n означает, что индекс первого необработанного слова всё ещё находится внутри списка.  
	# Каждая итерация этого цикла создаёт одну готовую строку.  
	# Внутри цикла мы сначала должны выяснить, какие слова поместятся в эту строку.  
        while i < n:  
            
	    # Заводим две новые переменные.  
	    # j - индекс следующего слова, которое проверяем. Если i = 0, то и j = 0.  
	    # И j начинает проверять слова с самого начала списка.  
	    # Когда слово успешно добавляется в текущую строку, j увеличивается на еденицу.  
            j = i  
	    # Суммарная длина слов.  
            # Здесь мы считаем только длину самих слов, без пробелов.  
	    # Например, если уже выбраны слова "This", "is" и "an", то line_lenght = 4 + 2 + 2, т.е. = 8  
            # Почему мы не учитываем пробелы сразу? Потому что кол-во пробелов между словами мы будем рассчитывать  
	    # отдельно, когда определим полный состав строки.  
            line_length = 0  
	    
	    # Выбираем слова для строки.  
            # Именно здесь алгоритм решает, сколько слов поместится в строку.  
	    # j < n - проверяем, существуют ли ещё слова в списке.  
	    # len(words[j]) - узнаём длину следующего слова.  
            # line_length - учитываем длину слов, которые уже выбраны.  
	    # (j - i) - учитываем пробелы, необходимые между словами.  
	    # Почему именно j - i ?  
            # Ширина строки: maxWidth = 16  
	    # В начале: i = 0, j = 0, line_length = 0, попытка добавить "This", длина слова - 4.  
            # Проверяем: 0 + 4 + (0 - 0) = 4. 4 не превышает 16. Слово помещается.  
	    # Выполняется: line_length += len(words[j])  
            # j += 1  
	    # Теперь: line_length = 4  
            # j = 1  
            # Выбранное слово: "This".  
	    # Попытка №2: добавить слово "is"  
            # Длина "is" - 2.  
            # Теперь j = 1, поэтому:  
	    # 4 + 2 + (1 - 0) = 7  
            # Почему прибавили еденицу? Потому что между "This" и "is" должен быть хотя бы один пробел.  
	    # 7 не превышает 16, поэтому слово добавляется.  
            # Теперь:  
            # line_length = 6  
            # j = 2  
            # Выбраны слова: "This", "is"  
            # Попытка №3: добавить "an"  
            # Длина "an" - 2  
            # Проверяем:  
            # 6 + 2 + (2 - 0) = 10  
            # Между тремя словами необходимо два промежутка, поэтому добавляется 2.  
            # Слово помещается.  
            # Теперь:  
            # line_length = 8  
            # j = 3  
            # Выбраны слова: "This", "is" "an"  
            # Попытка #4: добавить "example"  
            # Длина "example" - 7  
            # Проверяем:  
            # 8 + 7 + (3 - 0) = 18  
            # 18 больше 16! Следовательно, "example" уже не помещается в текущую строку.  
	    # Останавливаем цикл! Слово "example" не теряется. Мы просто не добавляем его  
            # в текущую строку. Оно остаётся для следующего прохода внешнего цикла.  
            # Алгоритм не удаляет слова из списка и не перемещает их физически. Он работает с индексами.  
            while j < n and line_length + len(words[j]) + (j - i) <= maxWidth:  
                line_length += len(words[j])  
                j += 1  
	    
            # Считаем количество слов и промежутков.  
            # После завершения внутреннего цикла выполняются строки ниже.  
            # В нашем примере i = 0; j = 3, поэтому:  
            # num_words = 3 - 0; num_words = 3  
	    # В строке три слова, а количество промежутков между словами:  
	    # num_gaps = 3 - 1; num_gaps = 2  
	    # Почему два гапа, а не три? Потому что промежуток находится между соседними словами.  
	    # Для четырёх слов было бы три промежутка. Для 1 слова - ноль промежутков.  
            num_words = j - i  
            num_gaps = num_words - 1  
	    
            # Проверяем, последняя ли это строка.  
            # Здесь проверяются два условия.  
	    # Условие #1: j == n  
  	    # Это означает, что все оставшиеся слова уже выбраны и больше слов в списке нет.  
	    # Значит, текущая строка - последняя.  
            # Условие #2: num_gaps == 0  
            # Это значит, что в строке только одно слово и у него нет промежутков, которые можно было бы выравнивать.  
	    # Поэтому его нужно просто расположить слева и подолнить пробелами справа.  
	    # Если выполняется хотя бы одно из этих улосвий, используем левое выравнивание.  
            if j == n or num_gaps == 0:  
                line = " ".join(words[i:j])  
                line += " " * (maxWidth - len(line))  
	
	    # Если строка не последняя и в ней больше одного слова,  
	    # необходимо распределить пробелы между словами равномерно.  
            # Сначала определяем, сколько всего пробелов должно быть в строке.  
	    # maxWidth - требуемая ширина строки.  
	    # line_length - суммарная длина всех слов без пробелов.  
	    # Разница между ними показывает, сколько пробелов нужно добавить.  
            else:  
                total_spaces = maxWidth - line_length  
	    # Определяем минимальное количество пробелов между соседними словами.  
	    # Оператор // выполняет целочисленное деление.  
            # Например: 10 // 3 = 3.  
            # Если у нас 10 пробелов и 3 промежутка, в каждый промежуток сначала помещаем по 3 пробела.  
                spaces_per_gap = total_spaces // num_gaps  
	    # Определяем, сколько пробелов осталось после равномерного распределения.  
	    # Оператор % возвращает остаток от деления.  
	    # Например: 10 % 3 = 1.  
	    # Значит, после распределения по 3 пробела в каждый промежуток останется ещё один пробел,  
	    # который нужно куда-то добавить.  
                extra_spaces = total_spaces % num_gaps  
	    
	    # Создаём пустой список частей будущей строки.  
	    # В него будем по очереди добавлять слова и промежутки из пробелов.  
            # Позже метод "".join(parts) объединит все эти части в одну строку.  
                parts = []  
	    
	    # Проходим по всем промежуткам между словами.  
	    # k - номер текущего промежутка, начиная с нуля.  
	    # Например, если num_pags = 2, цикл выполняется два раза:  
	    # сначала k = 0, затем k = 1.  
                for k in range(num_gaps):  

	    # Добавляем слово, которое находится перед текущим промежутком.  
	    # i + k - индекс этого слова в исходном списке words.  
	    # Например, если i = 0:  
	    # при k = 0 добавляем words[0],  
	    # при k = 1 добавляем words[1].  
                    parts.append(words[i + k])  
	    
	    # Назначаем текущему промежутку базовое количество пробелов.  
	    # Каждый промежуток изначально получает spaces_per_gap пробелов.  
                    spaces = spaces_per_gap  
	    # Проверяем, остались ли дополнительные пробелы,  
	    # которые нужно распределить между промежутками.  
	    # Если k меньше extra_spaces, добавляем к текущему промежутку ещё один пробел.  
	    # Это позволяет распределить лишние пробелы слева направо.  
                    if k < extra_spaces:  
                        spaces += 1  
	    
	    # Создаём строку из нужного количества пробелов.  
	    # Например, " " * 4 даст строку из четырёх пробелов.  
	    # Добавляем этот промежуток в список parts.  
                    parts.append(" " * spaces)  
	    
	    # После цикла все промежутки распределены.  
	    # Но последнее слово всё ещё не добавлено, потому что цикл  
	    # добавлял каждое слово только перед очередным промежутком.  
	    # Индекс j - 1 - индекс последнего слова текущей строки.  
                parts.append(words[j - 1])  

	    # Объединяем все слова и промежутки в одну готовую строку.  
	    # Пустая строка "" означает, что между частями не нужно добавлять  
	    # никаких доп. символов.  
	    # Все необходимые пробелы уже находятся внутри parts.  
                line = "".join(parts)  
	    
	    # Добавляем готовую строку в результирующий список.  
	    # К этому моменту строка уже имеет ширину maxWidth символов.  
            result.append(line)  
	    
	    # Перемещаем индекс i к первому слову, которое ещё не обработано.  
	    # Внутренний цикл остановился на индексе j:  
	    # слова от words[i] до words[j - 1] уже вошли в текущую строку.  
	    # Поэтому следующая строка должна начинаться с words[j].  
            # Присваивание i = j позволяет внешнему циклу продолжить работу  
	    # со следующего необработанного слова.  
            i = j  
	
	# Когда внешний цикл завершится, все слова будут распределены по строкам.  
	# Возвращаем готовый список result.  
	# Каждая строка имеет длину ровно maxWidth символов.  
        return result  
```
